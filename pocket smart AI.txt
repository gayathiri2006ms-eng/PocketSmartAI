<script>

let income = Number(localStorage.getItem("income")) || 0;
let budget = Number(localStorage.getItem("budget")) || 0;

let expenses =
    JSON.parse(localStorage.getItem("expenses")) || [];

let goal =
    JSON.parse(localStorage.getItem("goal")) || null;


// -----------------------------
// SAVE DATA
// -----------------------------

function saveData() {

    localStorage.setItem("income", income);

    localStorage.setItem("budget", budget);

    localStorage.setItem(
        "expenses",
        JSON.stringify(expenses)
    );

    localStorage.setItem(
        "goal",
        JSON.stringify(goal)
    );
}


// -----------------------------
// CURRENCY
// -----------------------------

function money(value) {

    return "₹" + Number(value).toLocaleString("en-IN");

}


// -----------------------------
// SET INCOME
// -----------------------------

function setIncome() {

    let value =
        Number(document.getElementById("incomeInput").value);

    if(value <= 0) {

        alert("Please enter a valid income.");

        return;
    }

    income = value;

    saveData();

    updateUI();

    document.getElementById("incomeInput").value = "";

}


// -----------------------------
// SET BUDGET
// -----------------------------

function setBudget() {

    let value =
        Number(document.getElementById("budgetInput").value);

    if(value <= 0) {

        alert("Please enter a valid budget.");

        return;
    }

    budget = value;

    saveData();

    updateUI();

    document.getElementById("budgetInput").value = "";

}


// -----------------------------
// ADD EXPENSE
// -----------------------------

function addExpense() {

    let name =
        document.getElementById("expenseName").value.trim();

    let amount =
        Number(document.getElementById("expenseAmount").value);

    let category =
        document.getElementById("expenseCategory").value;


    if(name === "" || amount <= 0) {

        alert("Please enter a valid expense.");

        return;
    }


    let expense = {

        id: Date.now(),

        name: name,

        amount: amount,

        category: category,

        date: new Date().toLocaleDateString("en-IN")

    };


    expenses.push(expense);

    saveData();

    updateUI();

    document.getElementById("expenseName").value = "";

    document.getElementById("expenseAmount").value = "";

}


// -----------------------------
// DELETE EXPENSE
// -----------------------------

function deleteExpense(id) {

    expenses =
        expenses.filter(expense => expense.id !== id);

    saveData();

    updateUI();

}


// -----------------------------
// TOTAL EXPENSE
// -----------------------------

function totalExpenses() {

    return expenses.reduce(
        (total, expense) =>
            total + Number(expense.amount),
        0
    );

}


// -----------------------------
// UPDATE DASHBOARD
// -----------------------------

function updateUI() {

    let total = totalExpenses();

    let balance = income - total;


    document.getElementById("incomeDisplay")
        .innerText = money(income);

    document.getElementById("expenseDisplay")
        .innerText = money(total);

    document.getElementById("balanceDisplay")
        .innerText = money(balance);

    document.getElementById("budgetDisplay")
        .innerText = money(budget);


    updateBudgetProgress();

    displayExpenses();

    createChart();

    displayGoal();

}


// -----------------------------
// BUDGET PROGRESS
// -----------------------------

function updateBudgetProgress() {

    let total = totalExpenses();

    let progress = 0;

    if(budget > 0) {

        progress =
            Math.min((total / budget) * 100, 100);

    }


    document.getElementById("budgetProgress")
        .style.width = progress + "%";


    let message =
        document.getElementById("budgetMessage");


    if(budget === 0) {

        message.innerText =
            "No budget set yet.";

        return;
    }


    if(total > budget) {

        message.innerText =
            "⚠️ You have exceeded your monthly budget.";

        message.style.color = "#dc2626";

    } else {

        let remaining = budget - total;

        message.innerText =
            "You have " +
            money(remaining) +
            " remaining in your budget.";

        message.style.color = "#16a34a";

    }

}


// -----------------------------
// DISPLAY EXPENSES
// -----------------------------

function displayExpenses() {

    let list =
        document.getElementById("expenseList");


    if(expenses.length === 0) {

        list.innerHTML =
            "<p>No expenses added yet.</p>";

        return;
    }


    list.innerHTML = "";


    expenses
        .slice()
        .reverse()
        .forEach(expense => {

            let item =
                document.createElement("div");

            item.className =
                "expense-item";


            item.innerHTML = `

                <div class="expense-info">

                    <strong>
                        ${escapeHTML(expense.name)}
                    </strong>

                    <small>
                        ${expense.category}
                        •
                        ${expense.date}
                    </small>

                </div>

                <div>

                    <strong style="color:#dc2626">
                        -${money(expense.amount)}
                    </strong>

                    <br>

                    <button
                        class="delete-btn"
                        onclick="deleteExpense(${expense.id})"
                    >
                        Delete
                    </button>

                </div>

            `;


            list.appendChild(item);

        });

}


// -----------------------------
// ESCAPE HTML
// -----------------------------

function escapeHTML(text) {

    let div = document.createElement("div");

    div.textContent = text;

    return div.innerHTML;

}


// -----------------------------
// AI RECOMMENDATION
// -----------------------------

function getRecommendation() {

    let total = totalExpenses();

    let balance = income - total;

    let message =
        document.getElementById("aiMessage");


    if(income === 0) {

        message.innerHTML =
            "💡 Start by entering your monthly income.";

        return;
    }


    if(total === 0) {

        message.innerHTML =
            "🌟 You haven't recorded any expenses yet. " +
            "Add your expenses so I can understand your spending habits.";

        return;
    }


    let percentage =
        income > 0
        ? (total / income) * 100
        : 0;


    let categoryTotals = {};


    expenses.forEach(expense => {

        if(!categoryTotals[expense.category]) {

            categoryTotals[expense.category] = 0;

        }

        categoryTotals[expense.category] +=
            Number(expense.amount);

    });


    let highestCategory = "";

    let highestAmount = 0;


    for(let category in categoryTotals) {

        if(categoryTotals[category] > highestAmount) {

            highestAmount =
                categoryTotals[category];

            highestCategory =
                category;

        }

    }


    let recommendation = "";


    if(budget > 0 && total > budget) {

        recommendation =
            `⚠️ You have exceeded your budget by
            ${money(total - budget)}.
            Consider reducing your spending in
            <strong>${highestCategory}</strong>.`;

    }

    else if(percentage > 80) {

        recommendation =
            `⚠️ You have already used
            ${percentage.toFixed(0)}% of your income.
            Try limiting non-essential spending.`;

    }

    else if(percentage > 60) {

        recommendation =
            `💡 You have used
            ${percentage.toFixed(0)}% of your income.
            Keep monitoring your spending,
            especially in <strong>${highestCategory}</strong>.`;

    }

    else {

        recommendation =
            `✅ Your current spending is within a
            reasonable range.
            Your highest spending category is
            <strong>${highestCategory}</strong>.
            Keep maintaining your budget!`;

    }


    if(balance > 0) {

        recommendation +=
            `<br><br>
            💰 Current available balance:
            <strong>${money(balance)}</strong>.`;

    }


    message.innerHTML =
        recommendation;

}


// -----------------------------
// CHART
// -----------------------------

function createChart() {

    let chart =
        document.getElementById("chart");


    chart.innerHTML = "";


    if(expenses.length === 0) {

        chart.innerHTML =
            "<p>No spending data available.</p>";

        return;
    }


    let categories = {};


    expenses.forEach(expense => {

        if(!categories[expense.category]) {

            categories[expense.category] = 0;

        }

        categories[expense.category] +=
            Number(expense.amount);

    });


    let values =
        Object.values(categories);

    let max =
        Math.max(...values);


    Object.entries(categories)
        .forEach(([category, amount]) => {

            let container =
                document.createElement("div");

            container.className =
                "bar-container";


            let height =
                (amount / max) * 140;


            container.innerHTML = `

                <div
                    class="bar"
                    style="height:${height}px"
                    title="${money(amount)}"
                ></div>

                <small>
                    ${category}
                </small>

            `;


            chart.appendChild(container);

        });

}


// -----------------------------
// SAVINGS GOAL
// -----------------------------

function setGoal() {

    let name =
        document.getElementById("goalName").value.trim();

    let amount =
        Number(document.getElementById("goalAmount").value);


    if(name === "" || amount <= 0) {

        alert("Please enter a valid goal.");

        return;
    }


    goal = {

        name: name,

        amount: amount

    };


    saveData();

    displayGoal();


    document.getElementById("goalName").value = "";

    document.getElementById("goalAmount").value = "";

}


// -----------------------------
// DISPLAY GOAL
// -----------------------------

function displayGoal() {

    let element =
        document.getElementById("goalDisplay");


    if(!goal) {

        element.innerText =
            "No savings goal set.";

        return;
    }


    let balance =
        Math.max(income - totalExpenses(), 0);


    let percentage =
        Math.min(
            (balance / goal.amount) * 100,
            100
        );


    element.innerHTML = `

        🎯 <strong>${escapeHTML(goal.name)}</strong>

        <br>

        Target:
        ${money(goal.amount)}

        <br>

        Available:
        ${money(balance)}

        <br><br>

        <div class="progress-container">

            <div
                class="progress"
                style="width:${percentage}%"
            ></div>

        </div>

        ${percentage.toFixed(0)}% of goal

    `;

}


// -----------------------------
// INITIALIZE
// -----------------------------

updateUI();

</script>
